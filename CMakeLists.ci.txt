# CMakeLists.ci.txt — GitHub Actions CI 专用
# 通过 vcpkg manifest 模式编译 OCCT，与本地构建一致的版本
cmake_minimum_required(VERSION 3.20)
project(blender2step LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# ====== 1. vcpkg 工具链（由 lukka/run-vcpkg 自动设置 VCPKG_ROOT） ======
if(DEFINED ENV{VCPKG_ROOT})
    set(CMAKE_TOOLCHAIN_FILE "$ENV{VCPKG_ROOT}/scripts/buildsystems/vcpkg.cmake"
        CACHE STRING "Vcpkg toolchain file")
endif()

# ====== 2. 通过 vcpkg manifest 模式查找依赖 ======
find_package(OpenCASCADE CONFIG REQUIRED)
message(STATUS "OpenCASCADE found: ${OpenCASCADE_INCLUDE_DIRS}")

# ====== 5. Python 路径 ======
if(NOT DEFINED PYTHON_INCLUDE_DIR)
    if(DEFINED ENV{PYTHON_INCLUDE_DIR})
        set(PYTHON_INCLUDE_DIR "$ENV{PYTHON_INCLUDE_DIR}" CACHE PATH "Python include dir")
    else()
        set(PYTHON_INCLUDE_DIR "C:/Python311/include" CACHE PATH "Python include dir")
    endif()
endif()
if(NOT DEFINED PYTHON_LIBRARY)
    if(DEFINED ENV{PYTHON_LIBRARY})
        set(PYTHON_LIBRARY "$ENV{PYTHON_LIBRARY}" CACHE FILEPATH "Python library")
    else()
        set(PYTHON_LIBRARY "C:/Python311/libs/python311.lib" CACHE FILEPATH "Python library")
    endif()
endif()

# ====== 6. 源文件 ======
include_directories(${PYTHON_INCLUDE_DIR} ${OpenCASCADE_INCLUDE_DIRS} ${CMAKE_CURRENT_SOURCE_DIR}/include)

add_library(step_exporter SHARED
    src/shape/fix_shape.cpp
    src/shape/create_shape_from_mesh.cpp
    src/shape/create_solid_from_mesh.cpp
    src/shape/shape_fix.cpp
    src/shape/rounded_box.cpp
    src/curve/curve.cpp
    src/curve/curve_dict.cpp
    src/curve/curve_poly.cpp
    src/curve/curve_bezier.cpp
    src/curve/curve_nurbs.cpp
    src/curve/curve_common.cpp
    src/export/export_scene.cpp
    src/export/enhanced_parse.cpp
    src/export/enhanced_progress.cpp
    src/export/enhanced_writer.cpp
    src/export/enhanced_objects.cpp
    src/export/enhanced_transfer.cpp
    src/export/enhanced.cpp
    src/export/incremental.cpp
    src/export/module.cpp
    src/step_converter.cpp
    src/cylinder/cylinder_detector.cpp
    src/cylinder/cylinder_face_analyzer.cpp
    src/cylinder/cylinder_cluster_detector.cpp
    src/cylinder/cylinder_deduplicator.cpp
    src/cylinder/cylinder_geometry.cpp
    src/cylinder/cylinder_utils.cpp
    src/cylinder/cylinder_reconstruct.cpp
    src/cylinder/cylinder_parametric.cpp
)

include_directories(${PYTHON_INCLUDE_DIR} ${CMAKE_CURRENT_SOURCE_DIR}/include ${CMAKE_CURRENT_SOURCE_DIR}/src/cylinder)

set_target_properties(step_exporter PROPERTIES
    OUTPUT_NAME "_step_exporter"
    PREFIX ""
    SUFFIX ".pyd"
)

add_custom_command(TARGET step_exporter POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "$<TARGET_FILE:step_exporter>"
        "${CMAKE_SOURCE_DIR}/step_exporter/lib/_step_exporter.pyd"
    COMMENT "Copying _step_exporter.pyd to step_exporter/lib/"
)

# ====== 7. 链接 ======
target_link_libraries(step_exporter
    PRIVATE
        ${OpenCASCADE_LIBRARIES}
        ${PYTHON_LIBRARY}
        TKMesh TKFeat
    PUBLIC
        kernel32.lib user32.lib gdi32.lib
        winspool.lib shell32.lib ole32.lib oleaut32.lib
        uuid.lib comdlg32.lib advapi32.lib ws2_32.lib
)

if(MSVC)
    target_compile_options(step_exporter PRIVATE /W3 /EHsc /MD /utf-8)
    target_compile_definitions(step_exporter PRIVATE
        -D_CRT_SECURE_NO_WARNINGS
        -D_SILENCE_ALL_CXX17_DEPRECATION_WARNINGS
    )
endif()
